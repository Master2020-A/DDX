#
# 2. Para GPU con CUDA:
#    pip install torch torchvision --index-url https://download.pytorch.org/whl/cu118
#
# 3. TotalSegmentator descargará modelos (~1.5GB) al primer uso
#
# 4. nnU-Net requiere configuración de variables de entorno:
#    export nnUNet_raw="path/to/nnUNet_raw"
#    export nnUNet_preprocessed="path/to/nnUNet_preprocessed"
#    export nnUNet_results="path/to/nnUNet_results"
#
# 5. Para usar MONAI Label con 3D Slicer:
#    pip install monailabel
#    monailabel start_server --app path/to/app --studies path/to/studies
#
# 6. PaddlePaddle requiere versión específica según plataforma:
#    Windows CPU: paddlepaddle
#    Windows GPU: paddlepaddle-gpu
#    Linux: verificar documentación oficial
